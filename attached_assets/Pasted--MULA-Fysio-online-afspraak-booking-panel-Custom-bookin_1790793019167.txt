  class FluxApi {
    constructor(host, slug, token) {
      this.base = `${host.replace(/\/$/, "")}/api_schedule/v1/${encodeURIComponent(slug)}`;
      this.token = token;
    }
    async req(path, opts = {}) {
      const res = await fetch(this.base + path, {
        method: opts.method || "GET",
        headers: Object.assign({ Accept: "application/json", Authorization: `Bearer ${this.token || ""}` },
          opts.body ? { "Content-Type": "application/json" } : {}),
        body: opts.body ? JSON.stringify(opts.body) : undefined,
      });
      let data = null;
      try { data = await res.json(); } catch (_) { /* empty body */ }
      if (!res.ok) { const e = new Error((data && data.message) || `HTTP ${res.status}`); e.status = res.status; e.data = data; throw e; }
      return data;
    }
    q(params) {
      const u = new URLSearchParams();
      Object.entries(params).forEach(([k, v]) => { if (v) u.set(k, v); });
      const s = u.toString();
      return s ? `?${s}` : "";
    }
    config() { return this.req("/config"); }
    types() { return this.req("/appointment-types").then((r) => r.data); }
    locations(p) { return this.req("/locations" + this.q(p)).then((r) => r.data); }
    users(p) { return this.req("/users" + this.q(p)).then((r) => r.data.map((u) => Object.assign({}, u, { name: String(u.name).replace(/\s+/g, " ").trim() }))); }
    availability(p) { return this.req("/availability" + this.q(p)); }
    countries() { return this.req("/countries").then((r) => r.data); }
    book(body) { return this.req("/appointments", { method: "POST", body }); }
  }

        types = types.filter((t) => TYPE_TEXT[t.slug]).map((t) => Object.assign({}, t, TYPE_TEXT[t.slug][this.language])).sort((a, b) => rank(a) - rank(b));

        const results = await Promise.all(targets.map((u) =>
          this.api.availability({ appointment_type: s.type.slug, user: u.uuid, location: loc }).then((r) => ({ u, r }))));
        // merge into { date: [{t, user}] }

        dates.forEach((ds) => { const m = iso(mondayOf(ds)); if (!weeks.includes(m)) weeks.push(m); });
        const weekend = dates.some((ds) => { const g = parseDate(ds).getDay(); return (g === 0 || g === 6) && map[ds].length; });
        // start at the first week that has any free slot
        let first = weeks.findIndex((w) => dates.some((ds) => iso(mondayOf(ds)) === w && map[ds].length));

        s.week = Math.max(0, first);

        <div class="days" style="grid-template-columns:repeat(${n},1fr)">${days.map((ds) => {
          const c = (map[ds] || []).length; const d = parseDate(ds);
          return `<button class="day ${s.day === ds ? "sel" : ""}" data-a="day" data-id="${ds}" ${c ? "" : "disabled"}>${fmtWeekday(d, this.language)}<b>${d.getDate()}</b><small>${c ? c + " " + t.free : t.full}</small></button>`;
        }).join("")}</div>`;
      if (s.day && map[s.day]) {
        h += `<div class="label">${fmtDay(s.day, this.language)}</div><div class="slots">${map[s.day].map((x, i) => {
          const sel = s.slot && s.slot.t === x.t && s.slot.user.uuid === x.user.uuid;
          return `<button class="slot ${sel ? "sel" : ""}" data-a="slot" data-i="${i}">${esc(x.t)}${s.who || this.users.length === 1 ? "" : `<small>${esc(x.user.name.split(" ")[0])}</small>`}</button>`;
        }).join("")}</div>`;
      } else {
        h += `<p class="hint">${t.chooseDay}</p>`;
      }
      return h + `<button class="cta" data-a="next" ${s.slot ? "" : "disabled"}>${t.next} <span aria-hidden="true">→</span></button>`;
    }
    vForm() {
      const s = this.s, f = s.form, e = s.errors, t = this.t;
      const fld = (key, label, cls, attrs) => `<div class="f ${cls || ""} ${e[key] ? "bad" : ""}"><label for="${key}">${label}</label>
        <input id="${key}" data-f="${key}" value="${esc(f[key] || "")}" ${attrs || ""}>${e[key] ? `<span class="e">${esc(e[key])}</span>` : ""}</div>`;
      const loc = this.locations && this.locations.find((l) => l.uuid === s.location);
      return `<div class="rule"></div><h2>${t.almost}</h2>
        <div class="summary"><div class="t">${esc(s.type.name)}</div><div>${fmtDay(s.day, this.language)} ${t.at} ${esc(s.slot.t)} · ${esc(s.type.duration_minutes)} min</div><div>${t.with} ${esc(s.slot.user.name)}${loc && this.locations.length > 1 ? ", " + esc(loc.name) : ""}</div></div>
        ${s.alert ? `<div class="alert">${esc(s.alert)}</div>` : ""}
        <form class="form" novalidate>
          ${fld("first_names", t.firstName, "", 'autocomplete="given-name" maxlength="100"')}
          ${fld("surname_prefix", t.prefix, "s2", 'maxlength="50"')}
          ${fld("surname", t.surname, "s4", 'autocomplete="family-name" maxlength="100"')}
          ${fld("date_of_birth", t.birthDate, "", 'type="date" autocomplete="bday"')}
          <div class="f ${e.gender ? "bad" : ""}"><span class="lb">${t.gender}</span><div class="seg">${[["M", t.male], ["F", t.female], ["UN", t.other]].map(([v, l]) =>
            `<button type="button" class="${s.gender === v ? "sel" : ""}" data-a="g" data-v="${v}">${l}</button>`).join("")}</div>${e.gender ? `<span class="e">${esc(e.gender)}</span>` : ""}</div>
          ${fld("email_address", t.email, "", 'type="email" autocomplete="email" maxlength="200"')}
          ${fld("phone_number", t.phone, "", 'type="tel" autocomplete="tel" maxlength="30"')}
          ${fld("postal_code", t.postcode, "s4", 'autocomplete="postal-code"')}
          ${fld("house_number", t.houseNumber, "s2", 'maxlength="20"')}
          <div class="f ${e.description ? "bad" : ""}"><label for="description">${t.message}</label>
            <textarea id="description" data-f="description" rows="3" maxlength="200" placeholder="${t.messageExample}">${esc(f.description || "")}</textarea></div>
        </form>
        <button class="cta" data-a="submit" ${s.busy ? "disabled" : ""}>${s.busy ? t.sending : t.submit} <span aria-hidden="true">→</span></button>`;
    }
    vDone() {
      const s = this.s, t = this.t;
      const when = `${fmtDay(s.day, this.language)} ${t.at} ${s.slot.t}`;
      return `<div class="rule"></div><h2>${t.done}</h2>
        <p class="lead">${t.doneLead}</p>
        <div class="summary"><div class="t">${esc(s.type.name)}</div><div>${esc(when)}</div><div>${t.with} ${esc(s.slot.user.name)}</div></div>
        <div class="label">${t.goodToKnow}</div>
        <ul class="list">
          <li><span>01</span>${t.tip1}</li>
          <li><span>02</span>${t.tip2}</li>
          <li><span>03</span>${t.tip3} ${esc(this.email)}.</li>
        </ul>
        <button class="cta" data-a="finish">${t.finish} <span aria-hidden="true">→</span></button>`;
    }

    /* ---------------- events ---------------- */
    onInput(e) {
      const k = e.target.dataset && e.target.dataset.f;
      if (!k) return;
      this.s.form[k] = e.target.value;
      if (this.s.errors[k]) { delete this.s.errors[k]; e.target.closest(".f").classList.remove("bad"); const m = e.target.closest(".f").querySelector(".e"); if (m) m.remove(); }
    }
    onClick(e) {
      const b = e.target.closest("[data-a]");
      if (!b || b.disabled) return;
      const s = this.s, a = b.dataset.a;
      if (a === "retry") { this.boot = this.load(); this.render(); return; }
      if (a === "type") {
        s.type = this.types[+b.dataset.i]; s.step = 1; s.day = null; s.slot = null; s.alert = null;
        if (this.locations && this.locations.length === 1) s.location = this.locations[0].uuid;
        if (this.locations && this.locations.length === 0) s.location = null;
        this.users = null;
        if (!(this.locations && this.locations.length > 1) || s.location) this.loadAvailability(); else this.render();
        return;
      }
      if (a === "loc") { s.location = b.dataset.id; s.day = null; s.slot = null; this.loadAvailability(); return; }
      if (a === "reavail") { this.loadAvailability(); return; }
      if (a === "who") { s.who = b.dataset.id || null; s.day = null; s.slot = null; s.alert = null; this._keepScroll = true; this.loadAvailability(); return; }
      if (a === "wk") { s.week += +b.dataset.d; s.day = null; s.slot = null; this._keepScroll = true; }
      else if (a === "day") { s.day = b.dataset.id; s.slot = null; this._keepScroll = true; }
      else if (a === "slot") { s.slot = this.avail.map[s.day][+b.dataset.i]; this._keepScroll = true; }
      else if (a === "next") { s.step = 2; s.alert = null; }
      else if (a === "g") { s.gender = b.dataset.v; delete s.errors.gender; this._keepScroll = true; }
      else if (a === "submit") { this.submit(); return; }
      else if (a === "finish") { this.close(); return; }
      this.render();
    }

    validate() {
      const f = this.s.form, e = {}, t = this.t;
      const req = { first_names: t.requiredFirstName, surname: t.requiredSurname, date_of_birth: t.requiredBirthDate, email_address: t.requiredEmail, phone_number: t.requiredPhone };
      Object.entries(req).forEach(([k, m]) => { if (!String(f[k] || "").trim()) e[k] = m; });
      if (!e.email_address && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(f.email_address)) e.email_address = t.invalidEmail;
      if (!e.date_of_birth) {
        const d = new Date(f.date_of_birth); const now = new Date();
        if (isNaN(d) || d > now || d.getFullYear() < now.getFullYear() - 120) e.date_of_birth = t.invalidBirthDate;
      }
      if (!this.s.gender) e.gender = t.requiredGender;
      return e;
    }

    async submit() {
      const s = this.s;
      s.errors = this.validate(); s.alert = null;
      if (Object.keys(s.errors).length) { this.render(); this.focusFirstError(); return; }
      s.busy = true; this.render();
      const f = s.form, clean = (v) => (v && String(v).trim()) || null;
      const body = {
        appointment_type: s.type.slug, user: s.slot.user.uuid, location: s.location || null,
        date: s.day, timeslot: s.slot.t,
        contact: {
          first_names: clean(f.first_names), surname_prefix: clean(f.surname_prefix), surname: clean(f.surname),
          gender: s.gender, date_of_birth: f.date_of_birth, email_address: clean(f.email_address),
          phone_number: clean(f.phone_number), postal_code: clean(f.postal_code), house_number: clean(f.house_number),
          country_id: f.country_id || this.countryId || 159, description: clean(f.description),
        },
      };
      this.dispatchEvent(new CustomEvent("mula:submit", { bubbles: true, composed: true }));
      try {
        const r = await this.api.book(body);
        s.busy = false; s.result = r;
        this.dispatchEvent(new CustomEvent("mula:booked", { bubbles: true, composed: true, detail: { appointment: r.appointment } }));
        if (r.redirect_url) { window.location.href = r.redirect_url; return; }
        this.render();
      } catch (err) {
        s.busy = false;
        if (err.status === 409) {
          s.step = 1; s.slot = null; s.day = null;
          s.alert = this.t.taken;
          this.loadAvailability(); return;
        }
        if (err.status === 422 && err.data && err.data.errors) {
          const map = {};
          Object.entries(err.data.errors).forEach(([k, v]) => { map[k.replace(/^contact\./, "")] = this.language === "en" ? this.t.fieldError : [].concat(v)[0]; });
          if (map.country_id && !map.postal_code) map.postal_code = map.country_id;
          s.errors = map;
          const known = ["first_names", "surname_prefix", "surname", "date_of_birth", "gender", "email_address", "phone_number", "postal_code", "house_number", "description"];
          if (!Object.keys(map).some((k) => known.includes(k))) s.alert = this.t.requestError + " " + this.email + ".";
          this.render(); this.focusFirstError(); return;
        }
        s.alert = this.t.submitError + " " + this.email + ".";
        this.render();
      }
    }
    focusFirstError() {
      const el = this.root.querySelector(".f.bad input, .f.bad button, .alert");
      if (el) { el.scrollIntoView({ block: "center" }); if (el.focus) el.focus(); }
    }
  }

  customElements.define("mula-afspraak", MulaAfspraak);
})();
